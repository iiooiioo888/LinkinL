LinkinL 安裝 / 更新

git clone https://github.com/iiooiioo888/LinkinL.git
cd LinkinL && chmod +x scripts/*.sh
./scripts/install.sh
./scripts/update.sh --npm
cp .env.linkinl.example .env.linkinl
